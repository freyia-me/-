    cards.forEach(function(el){el.hidden=!(c==='all'||el.dataset.care.split(' ').indexOf(c)>-1);});
  }
  segs.forEach(function(b){b.addEventListener('click',function(){setCare(b.dataset.care);});});
  var q=(location.search.match(/[?&]care=(pro|home)/)||[])[1];
  var page=location.pathname.replace(/\/+$/,'').split('/').pop(); /* reformage-pro, reformage-homecare… */
  setCare(q||(/home/.test(page)?'home':/pro/.test(page)?'pro':'all'));

  /* --- окно товара --- */
  function esc(s){var d=document.createElement('div');d.textContent=s;return d.innerHTML;}
  function fill(card){
    var pd=card.querySelector('.pd'), img=card.querySelector('.ph img');
    var ph=dlg.querySelector('.dlg-ph img'); ph.src=img.src; ph.alt=img.alt; ph.style.objectPosition=img.style.objectPosition||'';
    dlg.querySelector('.en').textContent=card.querySelector('.en').textContent;
    dlg.querySelector('h3').textContent=card.querySelector('h3').textContent;
    dlg.querySelector('.desc').textContent=card.querySelector('.bd p').textContent;
    dlg.querySelector('.meta').innerHTML='<span>'+esc(pd.dataset.vol)+'</span><span>'+esc(pd.dataset.careT)+'</span>';
    dlg.querySelector('dl').innerHTML=[].map.call(pd.querySelectorAll('[data-k]'),function(r){
      return '<dt>'+esc(r.dataset.k)+'</dt><dd>'+esc(r.textContent)+'</dd>';}).join('');
  }
  function open(id,push){
    var card=root.querySelector('.card[data-id="'+id+'"]'); if(!card) return;
    fill(card);
    if(!dlg.open){ if(dlg.showModal) dlg.showModal(); else dlg.setAttribute('open',''); }
    dlg.scrollTop=0;
    if(push && location.hash!=='#p-'+id) history.pushState({jdl:id},'','#p-'+id);
  }
  function close(){
    if(dlg.open){ if(dlg.close) dlg.close(); else dlg.removeAttribute('open'); }
  }
  root.addEventListener('click',function(e){
    var a=e.target.closest('a[href^="#p-"]'); if(!a) return;
    e.preventDefault(); open(a.getAttribute('href').slice(3),true);
  });
  dlg.querySelector('.x').addEventListener('click',function(){ if(/^#p-/.test(location.hash)) history.back(); else close(); });
  dlg.addEventListener('click',function(e){ if(e.target===dlg) dlg.querySelector('.x').click(); });
  dlg.addEventListener('cancel',function(e){ e.preventDefault(); dlg.querySelector('.x').click(); });
  /* кнопки окна ведут на формы Tilda (#getprice_main…) — закрываем окно, чтобы форма была видна */
  [].forEach.call(dlg.querySelectorAll('.btns a'),function(a){a.addEventListener('click',function(){
    if(/^#p-/.test(location.hash)) history.replaceState(null,'',location.pathname+location.search);
    close();
  });});
  function sync(){ var m=location.hash.match(/^#p-([\w-]+)/); if(m) open(m[1],false); else close(); }
  window.addEventListener('popstate',sync);
  window.addEventListener('hashchange',sync);
  sync();

  /* --- появление при прокрутке --- */
  var items=root.querySelectorAll('.rv');
  if(!('IntersectionObserver' in window)){[].forEach.call(items,function(el){el.classList.add('in');});return;}
  var io=new IntersectionObserver(function(es){es.forEach(function(e){if(e.isIntersecting){e.target.classList.add('in');io.unobserve(e.target);}});},{threshold:.12,rootMargin:'0px 0px -30px 0px'});
  [].forEach.call(items,function(el){io.observe(el);});
})();
